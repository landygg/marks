trigger bookmarks_ad CREATE TRIGGER bookmarks_ad AFTER DELETE ON bookmarks BEGIN INSERT INTO bookmarks_fts(bookmarks_fts, rowid, title, summary, tags, notes, description, author, site, content) VALUES ('delete', old.id, old.title, old.summary, old.tags, old.notes, old.description, old.author, old.site, old.content); END
trigger bookmarks_ai CREATE TRIGGER bookmarks_ai AFTER INSERT ON bookmarks BEGIN INSERT INTO bookmarks_fts(rowid, title, summary, tags, notes, description, author, site, content) VALUES (new.id, new.title, new.summary, new.tags, new.notes, new.description, new.author, new.site, new.content); END
trigger bookmarks_au CREATE TRIGGER bookmarks_au AFTER UPDATE ON bookmarks BEGIN INSERT INTO bookmarks_fts(bookmarks_fts, rowid, title, summary, tags, notes, description, author, site, content) VALUES ('delete', old.id, old.title, old.summary, old.tags, old.notes, old.description, old.author, old.site, old.content); INSERT INTO bookmarks_fts(rowid, title, summary, tags, notes, description, author, site, content) VALUES (new.id, new.title, new.summary, new.tags, new.notes, new.description, new.author, new.site, new.content); END
